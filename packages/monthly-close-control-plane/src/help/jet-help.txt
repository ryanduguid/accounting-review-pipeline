usage: close-control jet [-h] --transactions TRANSACTIONS --currency CURRENCY

options:
  -h, --help            show this help message and exit
  --transactions TRANSACTIONS
                        one tenant's canonical transaction CSV with every line
                        of each journal; TransactionID names the journal

  --period-end PERIOD_END
                        last day of the period, YYYY-MM-DD; a journal dated
                        later is post-closing
