usage: close-control review [-h] --current CURRENT --prior PRIOR
                            [--mapping MAPPING]
                            [--mapping-policy MAPPING_POLICY]
                            [--balance-policy BALANCE_POLICY]
                            [--subledger SUBLEDGER]
                            [--review-note REVIEW_NOTE]
                            [--equity-schedule EQUITY_SCHEDULE]
                            [--equity-currency EQUITY_CURRENCY]
                            [--equity-currency-evidence EQUITY_CURRENCY_EVIDENCE]
                            [--calculation-evidence PATH]
                            [--require-calculation LABEL] --output OUTPUT
                            [--absolute-threshold ABSOLUTE_THRESHOLD]
                            [--percentage-threshold PERCENTAGE_THRESHOLD]
                            [--reconciliation-tolerance RECONCILIATION_TOLERANCE]
                            [--gst-mapping GST_MAPPING]
                            [--gst-settlements GST_SETTLEMENTS]
                            [--gst-absolute-tolerance GST_ABSOLUTE_TOLERANCE]
                            [--gst-percentage-tolerance GST_PERCENTAGE_TOLERANCE]

options:
  -h, --help            show this help message and exit
  --current CURRENT     current-period canonical trial-balance CSV
  --prior PRIOR         prior-period canonical trial-balance CSV
  --mapping MAPPING     optional AccountID,ReviewGroup mapping CSV
  --mapping-policy MAPPING_POLICY
                        optional permitted Section,ReviewGroup pairs CSV;
                        requires --mapping
  --balance-policy BALANCE_POLICY
                        optional AccountID,ExpectedBalance,ExpectMovement CSV
  --subledger SUBLEDGER
                        optional Tenant,AccountID,SubledgerBalance CSV
  --review-note REVIEW_NOTE
                        optional human acknowledgement JSON
  --equity-schedule EQUITY_SCHEDULE
                        independent ledger-equity movement JSON
  --equity-currency EQUITY_CURRENCY
                        independently confirmed trial-balance currency
  --equity-currency-evidence EQUITY_CURRENCY_EVIDENCE
                        reference confirming both TB currencies
  --calculation-evidence PATH
                        optional calculation-evidence JSON file; repeat for
                        more than one
  --require-calculation LABEL
                        a calculation this close needs; its absence is an
                        exception, never a silent omission. Repeat for more
                        than one
  --output OUTPUT       directory for the generated review pack
  --absolute-threshold ABSOLUTE_THRESHOLD
                        minimum absolute YTD variance for review
  --percentage-threshold PERCENTAGE_THRESHOLD
                        minimum proportional YTD variance for review, as a
                        fraction from 0 to 1, e.g. 0.10 for 10%
  --reconciliation-tolerance RECONCILIATION_TOLERANCE
                        maximum permitted GL/subledger difference
  --gst-mapping GST_MAPPING
                        optional AccountID,GSTClass CSV; runs the GST
                        reasonableness control
  --gst-settlements GST_SETTLEMENTS
                        Reference,Amount CSV of BAS payments, refunds and
                        clearing journals in the GST accounts, each as its
                        net debit; required with --gst-mapping, header only
                        when there were none
  --gst-absolute-tolerance GST_ABSOLUTE_TOLERANCE
                        GST difference in dollars allowed before review,
                        added to the percentage tolerance; a review heuristic,
                        not a rounding bound
  --gst-percentage-tolerance GST_PERCENTAGE_TOLERANCE
                        GST difference allowed before review, as a fraction
                        of the expected GST on sales and purchases, e.g. 0.01
                        for 1%
